### Формулировка (α=+2)

### Код

# card_manager.py:267
def delete(self, card_id: str, soft_delete: bool = True) -> bool:
    """
    Удаление карточки

    По умолчанию: soft delete (архивирование, статус = ARCHIVED)
    Информация НЕ ТЕРЯЕТСЯ, только меняется статус
    """
    card = self.read(card_id)
    if not card:
        return False

    if soft_delete:
        # ✅ Закон сохранения: данные НЕ удаляются физически
        card.status = CardStatus.ARCHIVED
        card.modified = datetime.datetime.now()

        # Обновить файл вместо удаления
        file_path = self._get_file_path(card_id)
        if file_path and file_path.exists():
            file_path.write_text(card.to_markdown(), encoding='utf-8')

        return True
    else:
        # ⚠️ Жёсткое удаление — только явно запрошенное
        file_path = self._get_file_path(card_id)
        if file_path and file_path.exists():
            file_path.unlink()

        return True
```

# ✅ ХОРОШО: Единственный источник истины
class CardManager:
    def get_linked_cards(self, card_id: str) -> List[Card]:
        """Загружает связанные карточки по ID"""
        card = self.read(card_id)
        if not card:
            return []

        # Каждая карточка загружается из СВОЕГО файла (SSOT)
        linked = []
        for linked_id in card.linked_cards:
            linked_card = self.read(linked_id)  # Единственный источник!
            if linked_card:
                linked.append(linked_card)

        return linked

# ❌ ПЛОХО: Дублирование данных

# ✅ Закон когезии: Всё про задачу — в одном классе
@dataclass
class TaskCard(Card):
    """Карточка-задача: данные + логика вместе"""

    # Данные задачи
    due_date: Optional[datetime.datetime] = None
    reminder: Optional[datetime.datetime] = None
    estimated_duration: Optional[int] = None
    completion_criteria: List[str] = field(default_factory=list)

    # Логика задачи (в том же классе!)
    def is_overdue(self) -> bool:
        """Проверка просрочки — рядом с данными"""
        if not self.due_date:
            return False
        return datetime.datetime.now() > self.due_date

    def complete(self) -> None:
        """Завершение задачи — рядом с данными"""
        self.status = CardStatus.COMPLETED
        self.modified = datetime.datetime.now()

    def get_time_remaining(self) -> Optional[timedelta]:
        """Оставшееся время — рядом с данными"""
        if not self.due_date:
            return None
        return self.due_date - datetime.datetime.now()
```

# card.py:78
@staticmethod
def generate_id() -> str:
    """
    ✅ Закон предсказуемости: ID всегда формата card_YYYYMMDD_HHMMSS_xxxx
    Пользователь ЗНАЕТ, что увидит, как выглядит ID
    """
    now = datetime.datetime.now()
    timestamp = now.strftime("%Y%m%d_%H%M%S")
    rand_suffix = ''.join(random.choices(string.ascii_lowercase + string.digits, k=4))
    return f"card_{timestamp}_{rand_suffix}"

    # Никогда: 12345, UUID, случайная строка

# card_manager.py:305
def search(
    self,
    query: str = None,
    card_type: CardType = None,
    tags: List[str] = None,
    status: CardStatus = None,
    limit: int = 100
) -> List[Card]:
    """
    ✅ Закон предсказуемости:
    Одинаковые параметры → одинаковые результаты

    Детерминированный порядок:
    1. Применить фильтр query
    2. Применить фильтр card_type
    3. Применить фильтр tags
    4. Применить фильтр status
    5. Вернуть первые `limit` результатов

    НЕТ случайности, НЕТ зависимости от внешних факторов
    """
    all_cards = self._load_all_cards()

    if query:
        all_cards = [c for c in all_cards if query.lower() in c.title.lower()]
    if card_type:
        all_cards = [c for c in all_cards if c.type == card_type]
    if tags:
        all_cards = [c for c in all_cards if any(t in c.tags for t in tags)]
    if status:
        all_cards = [c for c in all_cards if c.status == status]

    return all_cards[:limit]  # Детерминированно
```

    ✅ Закон градуального усложнения:
    Простое использование: TaskCard.create_task("Задача")
    Сложное использование: TaskCard.create_task("Задача", due_date=..., priority=...)
    """
    now = datetime.datetime.now()
    card_id = cls.generate_id()

    return cls(
        id=card_id,
        type=CardType.TASK,
        title=title,
        created=now,
        modified=now,
        content=content,
        due_date=due_date,
        status=kwargs.get('status', CardStatus.PENDING),  # Default
        priority=kwargs.get('priority', Priority.MEDIUM),  # Default
        **kwargs
    )
```

# card_manager.py:211
def read(self, card_id: str) -> Optional[Card]:
    """
    ✅ Закон устойчивости: ошибки НЕ роняют систему
    """
    file_path = self._get_file_path(card_id)
    if not file_path or not file_path.exists():
        return None  # ✅ Нет файла — не ошибка, просто None

    try:
        content = file_path.read_text(encoding='utf-8')
        card = Card.from_markdown(content)
        self._add_to_cache(card)
        return card
    except (yaml.YAMLError, ValueError) as e:
        # ✅ Некорректный формат — логируем, возвращаем None
        print(f"⚠️  Ошибка загрузки {card_id}: {e}")
        return None
    except Exception as e:
        # ✅ Непредвиденная ошибка — логируем, НЕ падаем
        print(f"❌ Неожиданная ошибка: {e}")
        return None
