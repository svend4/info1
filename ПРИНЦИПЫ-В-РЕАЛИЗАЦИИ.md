```python
def _extract_alpha(self, content: str) -> Optional[int]:
    """Extract abstraction level from document metadata"""
    patterns = [
        r'Уровень абстракции:\s*α=([+-]?\d+)',
        r'Level of abstraction:\s*α=([+-]?\d+)',
        r'α=([+-]?\d+)'
    ]
    for pattern in patterns:
        match = re.search(pattern, content, re.IGNORECASE)
        if match:
            return int(match.group(1))
    return None
```

**CD (Connectivity Density):**
```python
def calculate_metrics(self):
    total_docs = len(self.docs)
    total_connections = sum(len(links) for links in self.connections.values())

    # CD = (actual / max_possible) * 100
    max_connections = total_docs * (total_docs - 1)
    cd = (total_connections / max_connections * 100) if max_connections > 0 else 0
```
> Трассировка: ФИЛОСОФИЯ → ЗАКОНЫ → МЕТОДОЛОГИЯ → Формула CD → Python

**VT (Vertical Traceability):**
```python
# Count vertical connections (different α levels)
vertical_connections = 0
for doc_id, links in self.connections.items():
    doc_alpha = self.docs[doc_id].get('alpha')
    for link in links:
        link_alpha = self.docs.get(link, {}).get('alpha')
        if doc_alpha is not None and link_alpha is not None:
            if doc_alpha != link_alpha:  # Different levels → vertical
                vertical_connections += 1

vt = (vertical_connections / total_connections * 100) if total_connections > 0 else 0
```

**DB (Directional Balance):**

```python
# DB = |⇑ - ⇓| / avg(⇑, ⇓) * 100
```

```python
def get_adaptive_thresholds(project_type: str) -> Dict:
    """Returns adaptive thresholds for project type"""
    if project_type not in ADAPTIVE_THRESHOLDS:
        return DEFAULT_THRESHOLDS

    adaptive = ADAPTIVE_THRESHOLDS[project_type].copy()
    # Merge with defaults for missing metrics
    for metric in DEFAULT_THRESHOLDS:
        if metric not in adaptive:
            adaptive[metric] = DEFAULT_THRESHOLDS[metric]

    return adaptive
```

```python
def generate_recommendations(self) -> List[str]:

    recommendations = []
    metrics = self.current_metrics

    # CD too high → reduce connections
    if metrics['cd'] > self.thresholds['cd']['max']:
```

```python
def calculate_correlation(self, metric1: str, metric2: str) -> float:
    """Calculate Pearson correlation between metrics"""
    values1 = [p['metrics'][metric1] for p in self.projects]
    values2 = [p['metrics'][metric2] for p in self.projects]

    # Pearson correlation coefficient
    mean1 = sum(values1) / n
    mean2 = sum(values2) / n
    cov = sum((values1[i] - mean1) * (values2[i] - mean2) for i in range(n)) / n
    std1 = math.sqrt(sum((v - mean1) ** 2 for v in values1) / n)
    std2 = math.sqrt(sum((v - mean2) ** 2 for v in values2) / n)
    correlation = cov / (std1 * std2)

    return correlation
```
