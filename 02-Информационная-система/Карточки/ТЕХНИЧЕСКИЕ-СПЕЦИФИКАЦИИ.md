# Технические спецификации карточной системы [α=-1]

## Введение

Этот документ содержит детальные технические спецификации для реализации карточной системы организации информации. Уровень абстракции α=-1 — конкретные технические решения для концепции (α=0).

### Структура YAML + Markdown

Каждая карточка — это файл с расширением `.card.md`, содержащий:
1. YAML frontmatter с метаданными
2. Markdown-тело с содержимым

```yaml
# Принять лекарство от давления

## Описание

Ежедневный приём лекарства от повышенного давления согласно назначению врача.

## Инструкция

1. Достать таблетку из упаковки в шкафчике (полка 2)
2. Принять с полным стаканом воды
3. Измерить давление через 30 минут
4. Записать результат в дневник

## Примечания

- Принимать строго в одно и то же время
- Не пропускать приём
- При давлении выше 160/100 — связаться с врачом

## История выполнения

- 2026-01-22 19:05: Выполнено. Давление 135/85
- 2026-01-21 19:00: Выполнено. Давление 140/90
- 2026-01-20 19:10: Выполнено с задержкой. Давление 138/88
```

## Система идентификаторов

### Формат ID

```
card_YYYYMMDD_HHMMSS_RAND

Где:
- card_ — префикс
- YYYYMMDD — дата создания (год, месяц, день)
- HHMMSS — время создания (час, минута, секунда)
- RAND — 4 случайных символа [a-z0-9] для уникальности

Пример: card_20260123_143052_a8f3
```

### Генерация ID (Python):

```python
import datetime
import random
import string

def generate_card_id():
    now = datetime.datetime.now()
    timestamp = now.strftime("%Y%m%d_%H%M%S")
    rand_suffix = ''.join(random.choices(string.ascii_lowercase + string.digits, k=4))
    return f"card_{timestamp}_{rand_suffix}"
```

## API для работы с карточками

### Основные операции (CRUD)

#### 1. Создание карточки

```python
def create_card(card_type: str, title: str, **kwargs) -> Card:
    """
    Создать новую карточку.

    Args:
        card_type: Тип карточки (task, medicine, recipe, и т.д.)
        title: Заголовок карточки
        **kwargs: Дополнительные поля в зависимости от типа

    Returns:
        Card: Созданная карточка
    """
    card_id = generate_card_id()
    now = datetime.datetime.now()

    card = Card(
        id=card_id,
        type=card_type,
        title=title,
        created=now,
        modified=now,
        **kwargs
    )

    card.save()
    return card
```

#### 2. Чтение карточки

```python
def read_card(card_id: str) -> Card:
```

#### 3. Обновление карточки

```python
def update_card(card_id: str, **updates) -> Card:
    """
    Обновить поля карточки.

    Args:
        card_id: Идентификатор карточки
        **updates: Поля для обновления

    Returns:
        Card: Обновлённая карточка
    """
    card = Card.load(card_id)

    for key, value in updates.items():
        setattr(card, key, value)

    card.modified = datetime.datetime.now()
    card.save()

    return card
```

#### 4. Удаление карточки

```python
def delete_card(card_id: str, soft_delete: bool = True) -> bool:
    """
    Удалить карточку.

    Args:
        card_id: Идентификатор карточки
        soft_delete: Если True, переместить в архив, иначе удалить физически

    Returns:
        bool: True если успешно
    """
    card = Card.load(card_id)

    if soft_delete:
        card.status = "archived"
        card.save()
    else:
        card.delete()

    return True
```

### Работа со связями

```python
def link_cards(
    card_id_from: str,
    card_id_to: str,
    link_type: str = "related",
    description: str = None
) -> bool:
    """
    Создать связь между карточками.

    Args:
        card_id_from: ID исходной карточки
        card_id_to: ID целевой карточки
        link_type: Тип связи (parent, child, related, и т.д.)
        description: Описание связи

    Returns:
        bool: True если успешно
    """
    card_from = Card.load(card_id_from)

    link = {
        "id": card_id_to,
        "type": link_type,
        "description": description
    }

    if "linked_cards" not in card_from:
        card_from.linked_cards = []

    card_from.linked_cards.append(link)
    card_from.save()

    return True

def get_linked_cards(card_id: str, link_type: str = None) -> list[Card]:
    """
    Получить связанные карточки.

    Args:
        card_id: ID карточки
        link_type: Фильтр по типу связи (опционально)

    Returns:
        list[Card]: Список связанных карточек
    """
    card = Card.load(card_id)
    linked_ids = []

    for link in card.linked_cards:
        if link_type is None or link["type"] == link_type:
            linked_ids.append(link["id"])

    return [Card.load(lid) for lid in linked_ids]
```

## Интеграция с роботами

### Назначение задач роботам

```python
def assign_task_to_robot(card_id: str, robot_id: str) -> bool:
    """
    Назначить задачу роботу.

    Args:
        card_id: ID карточки-задачи
        robot_id: ID робота

    Returns:
        bool: True если успешно
    """
    card = Card.load(card_id)

    if card.type != "task":
        raise ValueError("Можно назначать только карточки типа 'task'")

    card.assigned_robot = robot_id
    card.status = "assigned"
    card.save()

    # Отправить команду роботу
    robot = Robot.get(robot_id)
    robot.add_task(card)

    return True
```

```python
    Обновление статуса задачи роботом.

    Args:
        card_id: ID карточки-задачи
        robot_id: ID робота
        status: Новый статус (in_progress, completed, failed)
        progress: Прогресс в процентах (0-100)
        notes: Примечания робота

    Returns:
        bool: True если успешно
    """
    card = Card.load(card_id)

    if card.assigned_robot != robot_id:
        raise ValueError("Робот не назначен на эту задачу")

    card.status = status

    if progress is not None:
        card.metadata["progress"] = progress

    if notes:
        if "robot_notes" not in card.metadata:
            card.metadata["robot_notes"] = []
        card.metadata["robot_notes"].append({
            "timestamp": datetime.datetime.now(),
            "note": notes
        })

    card.save()

    # Уведомить пользователя если задача завершена
    if status == "completed":
        notify_user(f"Задача '{card.title}' выполнена роботом")

    return True
```

### Кэширование

```python
class CardCache:
    """LRU кэш для часто используемых карточек."""

    def __init__(self, max_size: int = 1000):
        self.cache = {}
        self.max_size = max_size
        self.access_order = []

    def get(self, card_id: str) -> Card:
        if card_id in self.cache:
            # Переместить в конец (недавно использованные)
            self.access_order.remove(card_id)
            self.access_order.append(card_id)
            return self.cache[card_id]
        return None

    def put(self, card: Card):
        if len(self.cache) >= self.max_size:
            # Удалить наименее недавно используемую
            lru_id = self.access_order.pop(0)
            del self.cache[lru_id]

        self.cache[card.id] = card
        self.access_order.append(card.id)
```

```python
def encrypt_sensitive_fields(card: Card) -> Card:
    """
    Зашифровать конфиденциальные поля карточки.

    Конфиденциальные поля:
    - Медицинская информация
    - Финансовые данные
    - Личные контакты
    """
    sensitive_fields = ["medication", "medical_info", "contact_info"]

    for field in sensitive_fields:
        if hasattr(card, field):
            encrypted_value = encrypt(getattr(card, field))
            setattr(card, field, encrypted_value)

    return card
```
