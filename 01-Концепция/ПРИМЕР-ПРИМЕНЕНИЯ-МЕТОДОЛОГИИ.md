def plan_tasks(tasks, robots, house_map, user_schedule, history):
    """
    Главная функция планирования
    """
    # Шаг 1: Фильтрация задач
    active_tasks = filter_by_time_and_conditions(tasks, user_schedule)

    # Шаг 2: Приоритизация
    sorted_tasks = prioritize_tasks(active_tasks)

    # Шаг 3: Оптимизация последовательности
    optimized_sequence = optimize_task_order(sorted_tasks, house_map)

    # Шаг 4: Распределение по роботам
    assignments = assign_to_robots(optimized_sequence, robots)

    # Шаг 5: Проверка конфликтов
    final_plan = resolve_conflicts(assignments, house_map)

    return final_plan

def prioritize_tasks(tasks):
    """
    Приоритизация задач по формуле
    """
    for task in tasks:
        # Формула приоритета:
        # P = Urgency × Importance × (1 - Completion)

        urgency = calculate_urgency(task.deadline)
        # urgency = 1 - (time_left / total_time)
        # Чем ближе дедлайн, тем выше urgency

        importance = task.priority_weight
        # high = 1.0, medium = 0.6, low = 0.3

        completion = task.progress
        # 0.0 = не начата, 1.0 = завершена

        task.priority_score = urgency * importance * (1 - completion)

    return sorted(tasks, key=lambda t: t.priority_score, reverse=True)

def optimize_task_order(tasks, house_map):
    """
    Оптимизация последовательности для минимизации
    времени перемещений (travelling salesman problem)
    """
    # Используем A* алгоритм

    # Создаём граф помещений
    rooms_graph = build_graph_from_map(house_map)

    # Находим оптимальный маршрут через все локации задач
    optimal_path = find_optimal_path(
        start="current_position",
        tasks_locations=[t.location for t in tasks],
        graph=rooms_graph,
        algorithm="A*"
    )

    # Переупорядочиваем задачи согласно маршруту
    ordered_tasks = reorder_by_path(tasks, optimal_path)

    return ordered_tasks

def assign_to_robots(tasks, robots):
    """
    Распределение задач между роботами
    (Hungarian algorithm для оптимального назначения)
    """
    # Создаём матрицу стоимостей
    # cost[i][j] = стоимость назначения задачи i роботу j

    cost_matrix = []
    for task in tasks:
        row = []
        for robot in robots:
            cost = calculate_assignment_cost(task, robot)
            row.append(cost)
        cost_matrix.append(row)

    # Решаем задачу о назначениях
    assignments = hungarian_algorithm(cost_matrix)

    return assignments

def calculate_assignment_cost(task, robot):
    """
    Вычисление стоимости назначения задачи роботу
    """
    # Факторы:

    # 1. Совместимость типа
    if robot.type != task.required_robot_type:
        return INFINITY  # Несовместим

    # 2. Расстояние до задачи
    distance = calculate_distance(
        robot.current_location,
        task.location
    )
    distance_cost = distance / robot.speed

    # 3. Уровень заряда
    battery_cost = 0
    if robot.battery_level < task.estimated_battery_usage:
        battery_cost = PENALTY  # Нужна подзарядка

    # 4. Загруженность робота
    workload_cost = len(robot.task_queue) * QUEUE_PENALTY

    # 5. Специализация (если робот лучше подходит)
    specialization_bonus = 0
    if robot.specialization == task.specific_requirements:
        specialization_bonus = -BONUS  # Отрицательная стоимость = бонус

    # Итоговая стоимость
    total_cost = (
        distance_cost +
        battery_cost +
        workload_cost +
        specialization_bonus
    )

    return total_cost

def resolve_conflicts(assignments, house_map):
    """
    Разрешение конфликтов (например, два робота не могут
    работать в одной маленькой комнате одновременно)
    """
    conflicts = detect_conflicts(assignments, house_map)

    for conflict in conflicts:
        # Стратегии разрешения:

        # 1. Последовательное выполнение
        if conflict.type == "same_room":
            reschedule_sequential(conflict.tasks)

        # 2. Переназначение одной из задач другому роботу
        elif conflict.type == "resource_contention":
            reassign_task(conflict.task_lower_priority)

        # 3. Разделение задачи на подзадачи
        elif conflict.type == "large_task":
            split_task(conflict.task)

    return conflict_free_assignments

```python
def handle_robot_failure(robot_id, task_id):
    """
    Обработка сбоя робота
    """
    # 1. Остановить задачу
    task = get_task(task_id)
    task.status = "paused"
    task.progress = get_current_progress(robot_id)

    # 2. Попытка самоосвобождения
    send_command(robot_id, "try_escape")
    wait(30)  # 30 секунд на попытку

    if robot.status == "stuck":
        # 3. Не получилось, переназначаем задачу

        # Найти альтернативного робота
        alternative_robot = find_available_robot(
            type=task.required_robot_type,
            exclude=[robot_id]
        )

        if alternative_robot:
            # Переназначить
            reassign_task(task, alternative_robot)
            notify_user(
                "Пылесос-1 застрял. Задачу продолжит Пылесос-2."
            )
        else:
            # Нет альтернативы, уведомить пользователя
            notify_user(
                "Пылесос-1 застрял под диваном в гостиной. "
                "Требуется помощь. Уборка приостановлена."
            )

            # Опционально: вызвать робота-помощника освободить
            if robot_helper_available():
                dispatch_helper(robot_id, "rescue")

```markdown
## ПРИНЦИП: Централизованная координация распределённой системы

### Формулировка

> В распределённой системе, состоящей из множества автономных
> агентов, наличие центрального координирующего элемента
> значительно повышает эффективность при выполнении
> взаимосвязанных задач.

### Суть принципа

**Децентрализация выполнения + Централизация планирования = Оптимум**

Система разделяется на два уровня:

1. **Координатор** (один)
   - Глобальное знание о системе
   - Планирование и оптимизация
   - Разрешение конфликтов
   - НЕ выполняет рутинные задачи

2. **Исполнители** (много)
   - Локальное выполнение задач
   - Автономность в рамках задания
   - Обратная связь координатору
   - Специализация на конкретных операциях

### Применимость

Этот принцип применим не только к робототехнике, но и к:

**1. Биологические системы:**
- Мозг (координатор) + Органы (исполнители)
- Королева пчёл/муравьёв + Рабочие особи
- Нервная система: мозг → спинной мозг → мышцы

**2. Социальные структуры:**
- Командир + Солдаты
- Дирижёр + Оркестр
- CEO + Департаменты компании
- Капитан + Команда корабля

**3. Технологические системы:**
- Сервер + Клиенты (клиент-серверная архитектура)
- Операционная система: ядро + процессы
- Умный дом: хаб + устройства
- Компьютер: CPU + Периферия

**4. Программное обеспечение:**
- Microservices: API Gateway + Сервисы
- Orchestration: Kubernetes master + Nodes
- Distributed computing: Master node + Worker nodes

### Преимущества централизованной координации

1. **Глобальная оптимизация**
   - Координатор видит всю картину
   - Может оптимизировать в масштабе всей системы
   - Предотвращает локальные оптимумы

2. **Разрешение конфликтов**
   - Центральный арбитр при конфликте ресурсов
   - Непротиворечивые решения

3. **Простота отдельных агентов**
   - Исполнители не нужны сложные (дешевле)
   - Не нужна координация между собой

4. **Обучение и адаптация**
   - Координатор накапливает знания
   - Улучшает стратегии на основе опыта
   - Исполнители остаются простыми

### Недостатки и ограничения

1. **Единая точка отказа (SPOF)**
   - Если координатор выходит из строя → система парализована
   - Решение: резервный координатор, автономный режим исполнителей

2. **Масштабируемость**
   - Координатор становится узким местом при росте системы
   - Решение: иерархическая координация (координатор → суб-координаторы)

3. **Задержки связи**
   - Исполнители зависят от команд координатора
   - Решение: автономность в рамках задачи, локальные решения

### Оптимальные условия применения

Принцип наиболее эффективен когда:

✓ Задачи взаимосвязаны (есть зависимости, конфликты ресурсов)
✓ Требуется глобальная оптимизация
✓ Количество агентов: средне (10-100)
✓ Стоимость координации < Выигрыш от оптимизации

Принцип менее эффективен когда:

✗ Задачи полностью независимы
✗ Критична скорость реакции (нет времени на связь с координатором)
✗ Очень много агентов (>1000) → слишком большая нагрузка
✗ Нестабильная связь между координатором и агентами

### Формула эффективности

```
E_central = (G_optimization - C_coordination) / C_total

где:
E_central = эффективность централизованной координации
G_optimization = выигрыш от глобальной оптимизации
C_coordination = стоимость координации (вычисления, связь)
C_total = общая стоимость системы
```

Система эффективна если E_central > E_decentral (децентрализованной)

### Примеры из проекта

1. **Робот-координатор + мини-роботы**
   - Координатор: планирует уборку всего дома
   - Исполнители: каждый убирает свою комнату
   - Выигрыш: нет дублирования, оптимальные маршруты

2. **Информационная система: Пользователь + Проекты + Карточки**
   - Пользователь (координатор): видит все проекты
   - Проекты: группы карточек
   - Карточки: атомарные задачи

3. **Умный дом: Хаб + Устройства**
   - Хаб: координирует все устройства
   - Устройства: выполняют свои функции
```

## ПРИНЦИП ВЫСШЕГО ПОРЯДКА: Иерархия как основа организации сложности

### Философская формулировка

> Любая сложная система, стремящаяся к эффективности,
> неизбежно развивает иерархическую структуру, где
> высшие уровни координируют низшие, а низшие
> обеспечивают выполнение.

### Исторические корни

**Платон (427-347 до н.э.) - "Государство":**
- Идеальное государство иерархично
- Философы-правители (разум) → Воины (воля) → Ремесленники (желания)
- Каждый класс выполняет свою роль

**Аристотель (384-322 до н.э.) - "Политика":**
- "Целое больше суммы частей"
- Организация по принципу "архэ" (правящего начала)

**Конфуций (551-479 до н.э.):**
- Принцип "Лǐ" (ритуал, порядок)
- Небо → Император → Чиновники → Народ
- Каждый знает своё место и роль

**Средневековая Европа - "Великая цепь бытия" (Great Chain of Being):**
```
Бог
 ↓
Ангелы
 ↓
Человек
 ↓
Животные
 ↓
Растения
 ↓
Минералы
```

Всё в мире упорядочено иерархически.

### Научное обоснование

**Герберт Саймон (1916-2001) - "Архитектура сложности" (1962):**

> "Сложные системы эволюционируют быстрее, если состоят
> из относительно стабильных промежуточных форм
> (иерархических модулей)"

**Закон необходимого разнообразия (Ashby, 1956):**
- Для управления сложной системой нужен контроллер
  не менее сложный
- Но: иерархия позволяет контроллеру быть проще
  за счёт делегирования

**Теория систем (von Bertalanffy):**
- Эмерджентность: целое > суммы частей
- Иерархические системы более устойчивы
- Уровни абстракции позволяют управлять сложностью

Каждый уровень координирует нижний и подчиняется высшему.

### Математическая модель иерархии

**Закон Парето (80/20):**
- 20% элементов создают 80% эффекта
- В иерархии: 20% верхних уровней координируют 80% системы

**Степенной закон (Power Law):**
```
P(k) ∝ k^(-α)

где:
P(k) = вероятность узла иметь k связей
α = показатель степени (обычно 2-3)
```

Иерархические сети следуют степенному закону.

**Формула сложности иерархической системы:**

```
C_hierarchy = Σ(C_level_i) + C_coordination

C_hierarchy << C_flat

где:
C_hierarchy = сложность иерархической системы
C_flat = сложность плоской системы (все связаны со всеми)
C_level_i = сложность уровня i
C_coordination = стоимость координации между уровнями
```

Иерархия **экспоненциально** снижает сложность!

### Универсальный принцип

**ЗАКОН ИЕРАРХИЧЕСКОЙ ОРГАНИЗАЦИИ:**

> Эффективность сложной системы E пропорциональна степени
> её иерархической организации H и обратно пропорциональна
> стоимости координации C:
>
> E ∝ H / C

Где H можно измерить как количество уровней и чёткость разделения.

### Практические следствия для проекта

1. **Документация должна быть иерархичной**
   - Не плоский список, а дерево
   - От общего к частному
   - От абстрактного к конкретному

2. **Роботы должны быть организованы иерархично**
   - Координатор → Исполнители → Устройства
   - Не peer-to-peer сеть

3. **Информация должна быть многоуровневой**
   - Принципы → Концепции → Детали → Примеры
   - Каждый находит свой уровень понимания

4. **Циклы должны быть вложенными**
   - День ⊂ Неделя ⊂ Месяц ⊂ Год
   - Синхронизация между уровнями

```markdown
## МЕТАПРИНЦИП: Закон организации сложности

### Формулировка на естественном языке

> Любая система, стремящаяся к выживанию и эффективности в
> сложной среде, неизбежно развивает иерархическую структуру
> с разделением функций между уровнями, где каждый уровень
> оперирует на своей шкале абстракции и времени.

### Математическая формулировка

```
S = {L₀, L₁, L₂, ..., Lₙ}

где:
S = сложная система
Lᵢ = уровень иерархии i
L₀ = базовый уровень (исполнители)
Lₙ = высший уровень (координатор)

Для эффективной системы:
∀ i: Lᵢ₊₁ координирует Lᵢ
∀ i: complexity(Lᵢ₊₁) < complexity(Lᵢ)
∀ i: timescale(Lᵢ₊₁) > timescale(Lᵢ)

Эффективность E(S):
E(S) = Π(Effectiveness(Lᵢ)) / Cost(Coordination)

Оптимум при:
dE/dn = 0  (оптимальное количество уровней)
```

### Связь с фундаментальными законами

**Термодинамика:**
- Второй закон: энтропия стремится к максимуму
- Иерархия - способ локального снижения энтропии
- Упорядоченная система более эффективна

**Теория информации:**
- Информация = снижение неопределённости
- Иерархия = эффективное сжатие информации
- Высшие уровни - сжатое представление низших

**Эволюция:**
- Естественный отбор благоприятствует иерархическим структурам
- Модульность → эволюционируемость
- Иерархия → устойчивость к изменениям

### Философская интерпретация

Это переформулировка древнего принципа **"Ordo ab chao"** (Порядок из хаоса):

Хаос (неорганизованная сложность) →
Иерархия (организованная сложность) →
Эффективность

Или восточный принцип **"Дао порождает одно, одно порождает два, два порождает три, три порождает все вещи"** (Дао Дэ Цзин):

Единое целое → Разделение → Иерархия → Многообразие форм

### Применимость

Этот закон применим к:
- Любым сложным системам (>100 элементов)
- Биологическим организмам
- Социальным структурам
- Технологическим системам
- Информационным структурам
- Самой Вселенной

**Исключения:**
- Простые системы (<10 элементов) - плоская структура эффективнее
- Полностью независимые агенты - координация не нужна
```
